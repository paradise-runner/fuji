A different kind of agent framework

Small enough to embed. Serious enough to scale.

Fuji is a pure, naked core for agentic work. It gives you the runtime, tools, sessions, and provider integrations you need without forcing an application framework around them.

Use it as a Go library, run it headlessly as a one-shot process, or build it into systems that need to operate across a fleet.

package main

import (
    "context"
    "log"

    "fuji/pkg/config"
    "fuji/pkg/session"
)

func main() {
    cfg := config.Config{
        Cwd:      ".",
        Provider: "anthropic",
        Model:    "claude-sonnet",
        ApiKey:   "your-api-key",
    }

    sess, err := session.New(cfg)
    if err != nil {
        log.Fatal(err)
    }
    defer sess.Close()

    sess.Prompt(
        context.Background(),
        "Analyze this repo",
    )
}
